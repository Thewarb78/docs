make a cert 

create kdb and import cert:

mkdir -p ~/mqkeys && cd ~/mqkeys
runmqakm -keydb -create -db key.kdb -pw '<pw>' -type cms -stash

# CA chain, one per cert, unique labels
runmqakm -cert -add -db key.kdb -stashed -label ca-root -file root.pem
runmqakm -cert -add -db key.kdb -stashed -label ca-issuing -file issuing.pem

# bundle your cert + key, then import
openssl pkcs12 -export -out client.p12 -inkey client.key -in client.cer \
  -certfile ca-chain.pem -name myapp-cert -passout pass:<p12pw>

runmqakm -cert -import -file client.p12 -pw '<p12pw>' -type pkcs12 \
  -target key.kdb -target_stashed -target_type cms -label myapp-cert

runmqakm -cert -list -db key.kdb -stashed
chmod 600 key.kdb key.sth
rm client.p12 client.key

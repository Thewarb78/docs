apiVersion: argoproj.io/v1alpha1
kind: Workflow
metadata:
  generateName: spark-roundtrip-
  namespace: your-argo-ns
spec:
  serviceAccountName: spark-driver
  entrypoint: roundtrip

  arguments:
    parameters:
      - name: app-name
        value: "s3-roundtrip"
      - name: bucket
        value: "s3a://my-test-bucket/scratch"
      - name: executor-memory
        value: "2g"
      - name: executor-cores
        value: "2"

  ttlStrategy:
    secondsAfterCompletion: 3600
    secondsAfterFailure: 86400

  templates:
    - name: roundtrip
      activeDeadlineSeconds: 900
      container:
        image: myreg/spark-pyspark-smoke:4.1.2-1
        command: ["python3", "/apps/s3_roundtrip.py"]
        env:
          - name: APP_NAME
            value: "{{workflow.parameters.app-name}}"
          - name: BUCKET
            value: "{{workflow.parameters.bucket}}"
          - name: EXECUTOR_MEMORY
            value: "{{workflow.parameters.executor-memory}}"
          - name: EXECUTOR_CORES
            value: "{{workflow.parameters.executor-cores}}"
          - name: RUN_LABEL
            value: "{{workflow.parameters.app-name}}-{{workflow.uid}}"
          - name: POD_IP
            valueFrom:
              fieldRef:
                fieldPath: status.podIP
          - name: SERVICE_ACCOUNT
            valueFrom:
              fieldRef:
                fieldPath: spec.serviceAccountName
        resources:
          requests:
            cpu: "1"
            memory: 2Gi
          limits:
            memory: 2Gi

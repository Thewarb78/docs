APP_DIR=/path/to/app      # where the python script lives
MQHOST=mqhost             # from ConnectionName
MQPORT=1414

OUT=~/mq_audit_$(hostname).txt
{
echo "=== HOST / USER ==="; hostname; date; whoami; lsb_release -ds 2>/dev/null

echo; echo "=== MQ ENV VARS (current shell) ==="
env | grep -iE 'MQ|SSL|CCDT' | sed -E 's/(PASS[^=]*=).*/\1****/I'

echo; echo "=== MQ VARS SET IN PROFILE/SYSTEMD FILES ==="
sudo grep -rniE 'MQSERVER|MQCHL|MQSSL|MQCLNTCF|MQCCDT' /etc/environment /etc/profile /etc/profile.d \
  ~/.bashrc ~/.profile /etc/systemd/system 2>/dev/null

echo; echo "=== KEY / CONFIG / CCDT FILES ==="
sudo find / -xdev \( -name "*.kdb" -o -name "*.sth" -o -name "*.p12" -o -name "*.pfx" \
  -o -name "mqclient.ini" -o -name ".mqclient.ini" -o -name "AMQCLCHL.TAB" -o -iname "*ccdt*" \) \
  -not -path "/proc/*" 2>/dev/null

echo; echo "=== CONTENTS OF ANY mqclient.ini ==="
for f in $(sudo find / -xdev \( -name mqclient.ini -o -name .mqclient.ini \) 2>/dev/null); do
  echo "--- $f"; sudo cat "$f"; done

echo; echo "=== CERTS IN ANY .kdb (needs .sth alongside) ==="
for f in $(sudo find / -xdev -name "*.kdb" 2>/dev/null); do
  echo "--- $f"; sudo runmqakm -cert -list -db "$f" -stashed 2>&1; done

echo; echo "=== MQ CLIENT VERSION ==="
dspmqver 2>/dev/null || /opt/mqm/bin/dspmqver 2>&1

echo; echo "=== PYTHON / PYMQI ==="
which -a python3; python3 --version
python3 -m pip show pymqi 2>&1
sudo find / -xdev -type d -iname "pymqi*" -path "*-packages*" 2>/dev/null

echo; echo "=== MQ SETTINGS IN APP CODE/CONFIG ==="
grep -rniE 'SSLCipherSpec|KeyRepository|CertificateLabel|SSLKEYR|MQSERVER|MQCHL|ccdt|kdb|connect_with_options|pymqi\.connect' \
  "$APP_DIR" --include=*.{py,ini,cfg,conf,yaml,yml,json,env,sh} 2>/dev/null

echo; echo "=== CLIENT ERROR LOG (last 60 lines) ==="
tail -60 /var/mqm/errors/AMQERR01.LOG 2>/dev/null
tail -60 ~/IBM/MQ/data/errors/AMQERR01.LOG 2>/dev/null

echo; echo "=== DOES THE SERVER SPEAK TLS? ==="
timeout 10 openssl s_client -connect "$MQHOST:$MQPORT" -showcerts </dev/null 2>&1 | \
  grep -E 'subject=|issuer=|Verify return|Protocol|Cipher|CONNECTED|errno' | head -20
} > "$OUT" 2>&1

echo "Saved to $OUT"

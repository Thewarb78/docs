<#
.SYNOPSIS
    Gets PasswordLastSet for a single AD account via a Global Catalog.
    Pure .NET / ADSI - no RSAT ActiveDirectory module required.

.EXAMPLE
    .\Get-PasswordLastSet.ps1 jbloggs
    .\Get-PasswordLastSet.ps1 -Identity jbloggs -Domain emea.corp.local -Verbose
    .\Get-PasswordLastSet.ps1 -Identity j.bloggs@corp.local -Server gc01.corp.local
#>
[CmdletBinding()]
param(
    [Parameter(Mandatory, Position = 0)]
    [Alias('SamAccountName', 'User')]
    [string]$Identity,

    [string]$Domain,

    [string]$Server,

    [ValidateSet(3268, 3269)]
    [int]$Port = 3268,

    [pscredential]$Credential
)

Add-Type -AssemblyName System.DirectoryServices -ErrorAction SilentlyContinue

# --- helpers ---------------------------------------------------------------
function New-Root {
    param([string]$Path)

    if ($Credential) {
        $auth = [System.DirectoryServices.AuthenticationTypes]::Secure
        if ($Port -eq 3269) {
            $auth = $auth -bor [System.DirectoryServices.AuthenticationTypes]::SecureSocketsLayer
        }
        New-Object System.DirectoryServices.DirectoryEntry(
            $Path,
            $Credential.UserName,
            $Credential.GetNetworkCredential().Password,
            $auth)
    }
    else {
        # Single-arg ctor = bind as the current user. Passing $null,$null to the
        # 4-arg overload triggers an anonymous bind, which AD refuses with
        # "An operations error occurred".
        New-Object System.DirectoryServices.DirectoryEntry($Path)
    }
}

function Test-Bind {
    param([System.DirectoryServices.DirectoryEntry]$Entry, [string]$What)
    try {
        $Entry.RefreshCache()
        [void]$Entry.Guid   # forces the native bind
    }
    catch {
        throw ("Bind to {0} failed: {1}`n" -f $What, $_.Exception.Message +
               "Check: 1) port $Port open, 2) target is actually a GC, " +
               "3) you are running as a domain principal (use -Credential if not).")
    }
}

function ConvertFrom-AdLargeInteger($Value) {
    if ($null -eq $Value)      { return $null }
    if ($Value -is [int64])    { return [int64]$Value }
    if ($Value -is [string])   { return [int64]$Value }
    $t    = $Value.GetType()
    $high = $t.InvokeMember('HighPart', 'GetProperty', $null, $Value, $null)
    $low  = $t.InvokeMember('LowPart',  'GetProperty', $null, $Value, $null)
    return (([int64]$high) -shl 32) -bor (([int64]$low) -band [int64]0xFFFFFFFF)
}

function Get-DomainFromDN([string]$Dn) {
    (($Dn -split '(?<!\\),') |
        Where-Object { $_ -match '^DC=' } |
        ForEach-Object { $_.Substring(3) }) -join '.'
}

# --- locate a GC -----------------------------------------------------------
if ($Server) {
    $gc = if ($Server -match ':\d+$') { $Server } else { '{0}:{1}' -f $Server, $Port }
}
else {
    try {
        if ($Domain) {
            $ctxArgs = @('Domain', $Domain)
            if ($Credential) {
                $ctxArgs += $Credential.UserName
                $ctxArgs += $Credential.GetNetworkCredential().Password
            }
            $ctx    = New-Object System.DirectoryServices.ActiveDirectory.DirectoryContext $ctxArgs
            $forest = [System.DirectoryServices.ActiveDirectory.Domain]::GetDomain($ctx).Forest
        }
        else {
            $forest = [System.DirectoryServices.ActiveDirectory.Forest]::GetCurrentForest()
        }
        $gcHost = $forest.FindGlobalCatalog().Name
    }
    catch {
        $fallback = if ($Domain) { $Domain } else { $env:USERDNSDOMAIN }
        if (-not $fallback) { throw "Could not locate a Global Catalog: $($_.Exception.Message)" }
        Write-Verbose "GC discovery failed ($($_.Exception.Message)), falling back to '$fallback'"
        $gcHost = $fallback
    }
    $gc = '{0}:{1}' -f $gcHost, $Port
}
Write-Verbose "GC: $gc"

# --- confirm we can bind, and that it really is a GC -----------------------
$rootDse = New-Root "LDAP://$gc/RootDSE"
try {
    Test-Bind -Entry $rootDse -What "LDAP://$gc/RootDSE"

    $caps = @($rootDse.Properties['supportedCapabilities'])
    if ($caps -notcontains '1.2.840.113556.1.4.1791') {
        Write-Warning "$gc did not advertise the GC capability OID. Results may be incomplete."
    }

    if ($Domain) {
        $baseDn = 'DC=' + ($Domain -split '\.' -join ',DC=')
    }
    else {
        $baseDn = [string]$rootDse.Properties['rootDomainNamingContext'][0]
        if (-not $baseDn) { throw "Could not read rootDomainNamingContext from $gc." }
    }
}
finally {
    if ($rootDse) { try { $rootDse.Dispose() } catch { } }
}
Write-Verbose "Base: $baseDn"

# --- escape and search -----------------------------------------------------
$esc = $Identity -replace '\\', '\5c' -replace '\*', '\2a' `
                 -replace '\(', '\28' -replace '\)', '\29' -replace '/', '\2f'

$filter = "(&(objectCategory=person)(objectClass=user)" +
          "(|(sAMAccountName=$esc)(userPrincipalName=$esc)(distinguishedName=$esc)))"
Write-Verbose "Filter: $filter"

$root     = $null
$searcher = $null

try {
    $root = New-Root "LDAP://$gc/$baseDn"
    Test-Bind -Entry $root -What "LDAP://$gc/$baseDn"

    $searcher = New-Object System.DirectoryServices.DirectorySearcher
    $searcher.SearchRoot      = $root
    $searcher.Filter          = $filter
    $searcher.SearchScope     = 'Subtree'
    $searcher.ReferralChasing = 'None'   # never chase referrals off a GC
    $searcher.SizeLimit       = 10       # single-account lookup, no paging needed
    $searcher.ClientTimeout   = [timespan]::FromSeconds(60)

    foreach ($p in 'sAMAccountName', 'displayName', 'userPrincipalName',
                   'distinguishedName', 'pwdLastSet', 'userAccountControl', 'whenCreated') {
        [void]$searcher.PropertiesToLoad.Add($p)
    }

    $results = @($searcher.FindAll())

    if ($results.Count -eq 0) {
        Write-Warning "No account matching '$Identity' found via $gc under $baseDn."
        return
    }
    if ($results.Count -gt 1) {
        Write-Warning "$($results.Count) matches for '$Identity' - returning all. Use -Domain to narrow."
    }

    foreach ($r in $results) {
        $p   = $r.Properties
        $dn  = [string]$p['distinguishedname'][0]
        $uac = [int]$p['useraccountcontrol'][0]
        $raw = if ($p['pwdlastset'].Count) { ConvertFrom-AdLargeInteger $p['pwdlastset'][0] } else { $null }

        if ($null -eq $raw -or $raw -eq 0) {
            $pls = $null; $age = $null
        }
        else {
            $pls = [datetime]::FromFileTimeUtc($raw).ToLocalTime()
            $age = [math]::Round(((Get-Date) - $pls).TotalDays, 1)
        }

        [pscustomobject]@{
            SamAccountName        = [string]$p['samaccountname'][0]
            DisplayName           = if ($p['displayname'].Count)       { [string]$p['displayname'][0] }       else { $null }
            UserPrincipalName     = if ($p['userprincipalname'].Count) { [string]$p['userprincipalname'][0] } else { $null }
            Domain                = Get-DomainFromDN $dn
            PasswordLastSet       = $pls
            PasswordAgeDays       = $age
            MustChangeAtNextLogon = ($raw -eq 0)
            PasswordNeverExpires  = [bool]($uac -band 0x10000)
            Enabled               = -not ($uac -band 0x0002)
            WhenCreated           = if ($p['whencreated'].Count) { [datetime]$p['whencreated'][0] } else { $null }
            DistinguishedName     = $dn
            QueriedVia            = $gc
        }
    }
}
finally {
    if ($searcher) { try { $searcher.Dispose() } catch { } }
    if ($root)     { try { $root.Dispose()     } catch { } }
}

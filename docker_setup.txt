
df = spark.createDataFrame([(1, "a"), (2, "b"), (3, "c")], ["id", "val"])

df.write.format("parquet").mode("overwrite").save("s3a://bucket/tmpWriteParquetFile")
print("Parquet write OK")

df2 = spark.read.format("parquet").load("s3a://bucket/tmpWriteParquetFile")
df2.show()
print("Parquet read OK")

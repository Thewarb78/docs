pipeline {
    agent { label 'kubectl' }

    environment {
        KUBECONFIG = "${WORKSPACE}/${shared.k8s_cluster}-kubeconfig"
    }

    stages {
        stage('Sync secrets') {
            steps {
                script {
                    def config = readYaml file: 'secrets/dev.yaml'

                    def k8s_creds = cyberarkSecret('<pinniped-object>')

                    maskPasswords(varPasswordPairs: [
                        [password: k8s_creds.Content]
                    ]) {
                        withEnv(["SECRET_1=${k8s_creds.UserName}",
                                 "SECRET_2=${k8s_creds.Content}"]) {

                            sh """
                                set +x
                                export PINNIPED_USERNAME="\${SECRET_1}"
                                export PINNIPED_PASSWORD="\${SECRET_2}"

                                generate-kubeconfig.sh -c ${shared.k8s_cluster} -n datahub -e \\
                                    --kubeconfig ${shared.k8s_cluster}-kubeconfig

                                kubectl get ns > /dev/null
                            """

                            config.secrets.each { secret ->
                                def creds = cyberarkSecret(secret.cyberark_id)
                                def userKey = secret.keys?.username ?: 'username'
                                def passKey = secret.keys?.password ?: 'password'

                                maskPasswords(varPasswordPairs: [
                                    [password: creds.Content]
                                ]) {
                                    withEnv(["CA_USER=${creds.UserName}",
                                             "CA_PASS=${creds.Content}",
                                             "NAMESPACES=${config.managed_namespaces.join(' ')}"]) {

                                        sh """
                                            set +x
                                            for ns in \${NAMESPACES}; do
                                                kubectl create secret generic ${secret.name} \\
                                                  --from-literal=${userKey}="\${CA_USER}" \\
                                                  --from-literal=${passKey}="\${CA_PASS}" \\
                                                  -n \${ns} \\
                                                  --dry-run=client -o yaml | kubectl apply -f -
                                            done
                                        """
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }
    }
}

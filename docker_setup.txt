apiVersion: v1
kind: ConfigMap
metadata:
  name: fluent-bit-config
data:
  fluent-bit.conf: |
    [SERVICE]
        Flush         5
        Daemon        Off
        Log_Level     info
        Parsers_File  parsers.conf
        HTTP_Server   On
        HTTP_Listen   0.0.0.0
        HTTP_Port     2020

    [INPUT]
        Name              tail
        Tag               kube.spark
        Path              /var/log/pods/spark_*/*/*.log
        Path_Key          filename
        DB                /var/lib/fluent-bit/tail.db
        Parser            cri
        Read_from_Head    Off
        Skip_Long_Lines   On
        Refresh_Interval  10
        Mem_Buf_Limit     50MB

    [INPUT]
        Name              tail
        Tag               kube.etl
        Path              /var/log/pods/etl_*/*/*.log
        Path_Key          filename
        DB                /var/lib/fluent-bit/tail.db
        Parser            cri
        Read_from_Head    Off
        Skip_Long_Lines   On
        Refresh_Interval  10
        Mem_Buf_Limit     50MB

    [FILTER]
        Name         parser
        Match        kube.*
        Key_Name     filename
        Parser       podpath
        Reserve_Data On

    [OUTPUT]
        Name                 opensearch
        Match                *
        Host                 <vip-hostname>
        Port                 443
        HTTP_User            ${OS_USER}
        HTTP_Passwd          ${OS_PASSWD}
        tls                  On
        tls.verify           On
        Logstash_Format      On
        Logstash_Prefix      pod-logs-dev
        Logstash_DateFormat  %Y.%m.%d
        Suppress_Type_Name   On
        Generate_ID          On
        Retry_Limit          3
        net.keepalive        On

  parsers.conf: |
    [PARSER]
        Name        cri
        Format      regex
        Regex       ^(?<time>[^ ]+) (?<stream>stdout|stderr) (?<logtag>[^ ]*) (?<log>.*)$
        Time_Key    time
        Time_Format %Y-%m-%dT%H:%M:%S.%L%z

    [PARSER]
        Name   podpath
        Format regex
        Regex  ^/var/log/pods/(?<namespace>[^_]+)_(?<pod>[^_]+)_[^/]+/(?<container>[^/]+)/

<#
.SYNOPSIS
    Gets PasswordLastSet for a single AD account via a Global Catalog.

.EXAMPLE
    .\Get-PasswordLastSet.ps1 jbloggs
    .\Get-PasswordLastSet.ps1 -Identity jbloggs -Domain emea.corp.local
    .\Get-PasswordLastSet.ps1 -Identity j.bloggs@corp.local -Server gc01.corp.local
#>
[CmdletBinding()]
param(
    [Parameter(Mandatory, Position = 0)]
    [Alias('SamAccountName', 'User')]
    [string]$Identity,

    # Scope the GC search to one domain partition. Omit = whole forest.
    [string]$Domain,

    # Explicit GC host. Append :port to override, e.g. gc01:3269
    [string]$Server,

    [ValidateSet(3268, 3269)]
    [int]$Port = 3268,

    [pscredential]$Credential
)

Import-Module ActiveDirectory -ErrorAction Stop

# --- locate a GC -----------------------------------------------------------
if ($Server) {
    $gc = if ($Server -match ':\d+$') { $Server } else { '{0}:{1}' -f $Server, $Port }
}
else {
    $discover = @{ Discover = $true; Service = 'GlobalCatalog'; ErrorAction = 'Stop' }
    if ($Domain) { $discover['DomainName'] = $Domain }
    $gcHost = @((Get-ADDomainController @discover).HostName)[0]
    if (-not $gcHost) { throw "No Global Catalog found$(if($Domain){" for $Domain"})." }
    $gc = '{0}:{1}' -f $gcHost, $Port
}
Write-Verbose "GC: $gc"

# --- search base: empty string = whole forest on a GC bind -----------------
$searchBase = if ($Domain) { 'DC=' + ($Domain -split '\.' -join ',DC=') } else { '' }
Write-Verbose "Base: '$searchBase'"

# --- escape and build the filter ------------------------------------------
$esc = $Identity -replace '\\', '\5c' -replace '\*', '\2a' `
                 -replace '\(', '\28' -replace '\)', '\29' -replace '/', '\2f'

$filter = "(&(objectCategory=person)(objectClass=user)" +
          "(|(sAMAccountName=$esc)(userPrincipalName=$esc)(distinguishedName=$esc)))"

$params = @{
    LDAPFilter  = $filter
    Server      = $gc
    SearchBase  = $searchBase
    SearchScope = 'Subtree'
    Properties  = 'sAMAccountName', 'displayName', 'userPrincipalName',
                  'pwdLastSet', 'userAccountControl', 'whenCreated'
    ErrorAction = 'Stop'
}
if ($Credential) { $params['Credential'] = $Credential }

$found = @(Get-ADUser @params)

if ($found.Count -eq 0) {
    Write-Warning "No account matching '$Identity' found via $gc."
    return
}
if ($found.Count -gt 1) {
    Write-Warning "$($found.Count) matches for '$Identity' across the forest - returning all. Use -Domain to narrow."
}

foreach ($u in $found) {
    $uac = [int]$u.userAccountControl
    $raw = $u.pwdLastSet

    if ($null -eq $raw -or [int64]$raw -eq 0) {
        $pls = $null; $age = $null
    }
    else {
        $pls = [datetime]::FromFileTimeUtc([int64]$raw).ToLocalTime()
        $age = [math]::Round(((Get-Date) - $pls).TotalDays, 1)
    }

    [pscustomobject]@{
        SamAccountName        = $u.sAMAccountName
        DisplayName           = $u.displayName
        UserPrincipalName     = $u.userPrincipalName
        Domain                = (($u.DistinguishedName -split '(?<!\\),') |
                                 Where-Object { $_ -match '^DC=' } |
                                 ForEach-Object { $_.Substring(3) }) -join '.'
        PasswordLastSet       = $pls
        PasswordAgeDays       = $age
        MustChangeAtNextLogon = ($null -ne $raw -and [int64]$raw -eq 0)
        PasswordNeverExpires  = [bool]($uac -band 0x10000)
        Enabled               = -not ($uac -band 0x0002)
        WhenCreated           = $u.whenCreated
        DistinguishedName     = $u.DistinguishedName
        QueriedVia            = $gc
    }
}

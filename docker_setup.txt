apiVersion: argoproj.io/v1alpha1
kind: WorkflowTemplate
metadata:
  name: spark-job
spec:
  entrypoint: run

  # Fixed per namespace - this template is copied into each Argo namespace
  # with the correct SA, ConfigMap and Secret names for that namespace.
  serviceAccountName: spark-driver

  # Pods stick around until the workflow is cleaned up: an hour on success,
  # a day on failure. No podGC, so logs stay available for that window.
  ttlStrategy:
    secondsAfterCompletion: 3600
    secondsAfterFailure: 86400

  arguments:
    parameters:
      # --- required: only the dev knows these ---
      - name: app-name
      - name: bucket
      - name: script

      # --- optional: platform defaults, overridable ---
      # Driver heap is 2g from spark-defaults. Pod memory must exceed it -
      # JVM non-heap, the Python process and the Argo sidecar live in the gap.
      # Raising pod memory alone won't give the JVM more heap; set
      # SPARK_DRIVER_MEMORY too if a job needs it.
      - name: driver-pod-memory
        value: "3Gi"
        enum: ["3Gi", "4Gi", "6Gi", "8Gi"]
      - name: driver-cpu
        value: "500m"
        enum: ["500m", "1", "2"]
      - name: executor-memory
        value: "2g"
        enum: ["1g", "2g", "4g"]
      - name: executor-cores
        value: "2"
        enum: ["1", "2", "4"]

  templates:
    - name: run
      activeDeadlineSeconds: 900

      # resources can't be templated inline - Quantity fields are schema
      # validated at save time, before parameters are substituted.
      # podSpecPatch is a plain string, applied at pod creation.
      # "main" is the container name Argo assigns; it must match exactly.
      podSpecPatch: |
        containers:
          - name: main
            resources:
              requests:
                cpu: "{{workflow.parameters.driver-cpu}}"
                memory: "{{workflow.parameters.driver-pod-memory}}"
              limits:
                memory: "{{workflow.parameters.driver-pod-memory}}"

      # Driver mounts only. Executors are created by Spark, not Argo -
      # their mounts come from spark.kubernetes.executor.* config.
      volumes:
        - name: spark-conf
          configMap:
            name: spark-defaults-4-1-2
        - name: s3-creds
          secret:
            secretName: s3-credentials

      container:
        # Fixed: image tag pairs with the ConfigMap above.
        image: myreg/spark-pyspark-base:4.1.2
        imagePullPolicy: Always
        command: ["python3", "{{workflow.parameters.script}}"]
        env:
          - name: SPARK_CONF_DIR
            value: /opt/spark/conf
          - name: APP_NAME
            value: "{{workflow.parameters.app-name}}"
          - name: BUCKET
            value: "{{workflow.parameters.bucket}}"
          - name: EXECUTOR_MEMORY
            value: "{{workflow.parameters.executor-memory}}"
          - name: EXECUTOR_CORES
            value: "{{workflow.parameters.executor-cores}}"
          - name: RUN_LABEL
            value: "{{workflow.parameters.app-name}}-{{workflow.uid}}"
          - name: POD_IP
            valueFrom:
              fieldRef:
                fieldPath: status.podIP
        volumeMounts:
          - name: spark-conf
            mountPath: /opt/spark/conf
          - name: s3-creds
            mountPath: /etc/secrets
            readOnly: true

        # Static fallback - podSpecPatch overrides these at runtime.
        resources:
          requests:
            cpu: "500m"
            memory: "3Gi"
          limits:
            memory: "3Gi"
